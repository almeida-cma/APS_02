def button_verificar_click(self, **event_args):
    
    # =========================
    # ENTRADA DE DADOS
    # =========================
    
    nota_texto = self.text_box_nota.text
    frequencia_texto = self.text_box_freq.text
    
    
    # =========================
    # VALIDAÇÃO (campos vazios)
    # =========================
    
    if nota_texto == "" or frequencia_texto == "":
        self.label_resultado.text = "⚠️ Preencha todos os campos!"
        self.label_resultado.role = None
        return
    
    
    # =========================
    # CONVERSÃO PARA NÚMERO
    # =========================
    
    try:
        nota = float(nota_texto)
        frequencia = float(frequencia_texto)
    except:
        self.label_resultado.text = "❌ Digite apenas números!"
        self.label_resultado.role = None
        return
    
    
    # =========================
    # VALIDAÇÃO DOS LIMITES
    # =========================
    
    if nota < 0 or nota > 10:
        self.label_resultado.text = "❌ Nota inválida! Digite um valor entre 0 e 10."
        self.label_resultado.role = None
        return
    
    if frequencia < 0 or frequencia > 100:
        self.label_resultado.text = "❌ Frequência inválida! Digite um valor entre 0 e 100."
        self.label_resultado.role = None
        return
    
    
    # =========================
    # REGRA DE NEGÓCIO
    # É APROVADO? (AND = ambas condições verdadeiras)
    # =========================
    
    if nota >= 7 and frequencia >= 75:
        situacao = "✅ DIPLOMA CONCEDIDO!"
        cor = "success"
    else:
        situacao = "❌ NÃO ATENDE AOS REQUISITOS"
        cor = "danger"
        
        # Mensagem adicional explicando o motivo
        if nota < 7 and frequencia < 75:
            motivo = "Nota abaixo de 7 E frequência abaixo de 75%"
        elif nota < 7:
            motivo = "Nota abaixo de 7"
        else:
            motivo = "Frequência abaixo de 75%"
        
        situacao = f"❌ NÃO ATENDE AOS REQUISITOS\n({motivo})"
    
    
    # =========================
    # SAÍDA
    # =========================
    
    self.label_resultado.text = situacao
    self.label_resultado.role = cor
